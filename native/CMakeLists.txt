cmake_minimum_required(VERSION 3.13)
project(rdpt_bridge C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(FreeRDP 3 REQUIRED)
find_package(FreeRDP-Client 3 REQUIRED)
find_package(WinPR 3 REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(rdpt-bridge bridge.c)
target_link_libraries(rdpt-bridge PRIVATE freerdp-client freerdp winpr OpenSSL::Crypto)
if(MINGW)
  target_link_libraries(rdpt-bridge PRIVATE ws2_32)
  target_link_options(rdpt-bridge PRIVATE -static-libgcc)
endif()
if(NOT WIN32)
  target_compile_definitions(rdpt-bridge PRIVATE _GNU_SOURCE)
endif()
target_compile_options(rdpt-bridge PRIVATE -Wall -Wextra)
install(TARGETS rdpt-bridge RUNTIME DESTINATION bin)
